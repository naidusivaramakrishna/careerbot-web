from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
import uuid
import logging

logger = logging.getLogger(__name__)

class CorrelationTraceMiddleware(BaseHTTPMiddleware):
    """Middleware handling both correlation_id and trace_id"""
    
    async def dispatch(self, request: Request, call_next):
        # 1. Extract correlation_id from header (set by frontend)
        correlation_id = request.headers.get("X-Correlation-ID")
        
        if not correlation_id:
            # Generate if frontend didn't provide
            correlation_id = f"correlation_{uuid.uuid4().hex[:16]}"
            logger.warning(f"No correlation_id provided, generated: {correlation_id}")
        
        # 2. Extract parent_trace_id (if this is a downstream call)
        parent_trace_id = request.headers.get("X-Trace-ID")
        
        # 3. Generate NEW trace_id for THIS service request
        trace_id = f"backend_trace_{uuid.uuid4().hex[:12]}"
        
        # 4. Store in request state (accessible in route handlers)
        request.state.correlation_id = correlation_id
        request.state.trace_id = trace_id
        request.state.parent_trace_id = parent_trace_id
        
        # 5. Log request with BOTH IDs
        logger.info(
            f"[REQUEST_START] {request.method} {request.url.path}",
            extra={
                "correlation_id": correlation_id,
                "trace_id": trace_id,
                "parent_trace_id": parent_trace_id,
                "method": request.method,
                "path": request.url.path,
                "service": "backend"
            }
        )
        
        # 6. Process request
        import time
        start_time = time.time()
        
        try:
            response = await call_next(request)
            duration_ms = (time.time() - start_time) * 1000
            
            # 7. Log response with BOTH IDs
            logger.info(
                f"[REQUEST_END] {response.status_code} {request.url.path} ({duration_ms:.2f}ms)",
                extra={
                    "correlation_id": correlation_id,
                    "trace_id": trace_id,
                    "status_code": response.status_code,
                    "duration_ms": round(duration_ms, 2),
                    "service": "backend"
                }
            )
            
            # 8. Add BOTH IDs to response headers
            response.headers["X-Correlation-ID"] = correlation_id
            response.headers["X-Trace-ID"] = trace_id
            
            return response
            
        except Exception as e:
            duration_ms = (time.time() - start_time) * 1000
            
            logger.error(
                f"[REQUEST_ERROR] {request.url.path} ({duration_ms:.2f}ms): {e}",
                extra={
                    "correlation_id": correlation_id,
                    "trace_id": trace_id,
                    "error": str(e),
                    "service": "backend"
                },
                exc_info=True
            )
            raise

import httpx
import logging
from fastapi import Request

logger = logging.getLogger(__name__)

async def parse_resume(request: Request, file_path: str):
    """Parse resume by calling AI Layer"""
    
    # 1. Get IDs from request state (set by middleware)
    correlation_id = request.state.correlation_id
    trace_id = request.state.trace_id
    
    logger.info(
        f"[BACKEND_SERVICE] Starting resume parse",
        extra={
            "correlation_id": correlation_id,
            "trace_id": trace_id,
            "operation": "parse_resume",
            "file_path": file_path,
            "service": "backend"
        }
    )
    
    # 2. Call AI Layer with BOTH IDs
    ai_layer_url = "http://ai-layer:8001/api/v1/resume/extract"
    
    async with httpx.AsyncClient() as client:
        logger.info(
            f"[BACKEND_TO_AI] Calling AI Layer",
            extra={
                "correlation_id": correlation_id,
                "trace_id": trace_id,
                "ai_endpoint": ai_layer_url,
                "service": "backend"
            }
        )
        
        # 3. Forward BOTH IDs in headers
        response = await client.post(
            ai_layer_url,
            json={"file_path": file_path},
            headers={
                "X-Correlation-ID": correlation_id,  # Same correlation
                "X-Trace-ID": trace_id              # Forward as parent
            }
        )
        
        # 4. Get child trace_id from AI Layer response
        ai_trace_id = response.headers.get("X-Trace-ID")
        
        logger.info(
            f"[AI_RESPONSE] Received from AI Layer",
            extra={
                "correlation_id": correlation_id,
                "trace_id": trace_id,              # Backend trace
                "ai_trace_id": ai_trace_id,        # AI Layer trace (child)
                "status_code": response.status_code,
                "service": "backend"
            }
        )
        
        result = response.json()
    
    # 5. Store in database with correlation_id
    logger.info(
        f"[DB_STORE] Storing parsed resume",
        extra={
            "correlation_id": correlation_id,
            "trace_id": trace_id,
            "operation": "store_resume",
            "service": "backend"
        }
    )
    
    # Store logic...
    
    return result

    responseData?.sessionId ||

    responseData?.session?.sessionId ||

    responseData?.result?.session_id ||
    responseData?.id ||
    responseData?.sessionId;
  // NO validation after this. If all undefined, code continues silently.

    const sessionId =
      responseData?.session_id ??
      responseData?.sessionId ??
      responseData?.session?.session_id ??
      responseData?.data?.session_id;

    if (!sessionId) {
      throw new Error(
        `[communicationApi.startSession] session_id missing in response. ` +
        `Response keys: ${Object.keys(responseData ?? {}).join(', ')}. ` +

    // In JSX:

      onClick={() => refetch()}

                 rounded-lg hover:opacity-90 transition-opacity"
    >
      Try Again
    </button>

[BUG-007]  BroadcastChannel created on every render — memory leak
----------------------------------------------------------------------
Priority  : P2
File      : src/contexts/DashboardContext.tsx
Line      : ~80-100 (inside component body, not useEffect)
Category  : Bug — Memory Leak, Duplicate Event Handlers

Problem:
  BroadcastChannel is instantiated inside the component function body,
  not inside a useEffect. This means a new BroadcastChannel is created
  on EVERY render. The old channels are never closed. Over the lifecycle
  of a typical session (many renders), hundreds of channels accumulate.
  When a credit update fires, every accumulated channel's onmessage
  handler triggers — meaning setCreditsRemaining is called hundreds of
  times in rapid succession, causing excessive re-renders of the entire
  context consumer tree.

Evidence:
  // WRONG — runs on every render:
  const channel = new BroadcastChannel('careerbot_credits');
  channel.onmessage = (event) => {
    if (mountedRef.current) {
      setCreditsRemaining(event.data.remaining);
    }
  };

Solution:
  Move entirely into useEffect with useRef:

    const channelRef = useRef<BroadcastChannel | null>(null);

    useEffect(() => {
      try {
        channelRef.current = new BroadcastChannel('careerbot_credits');
        channelRef.current.onmessage = (event) => {
          if (mountedRef.current) {
            setCreditsRemaining(event.data.remaining);
          }
        };
      } catch {
        // Safari / older browsers that don't support BroadcastChannel
        // Fall through to the window focus fallback below
      }

      return () => {
        channelRef.current?.close();
        channelRef.current = null;
      };
    }, []); // intentionally empty — run exactly once on mount

Impact:
  Without fix — progressive memory leak on long sessions. Users who
  stay logged in for hours experience increasing UI lag. Credit updates
  trigger dozens of state updates simultaneously.

[BUG-009]  useCreditCheck promise hangs forever if modal is dismissed
----------------------------------------------------------------------
Priority  : P2
File      : src/hooks/useCreditCheck.ts
Line      : ~120-160 (promise creation)
Category  : Bug — UI Freeze, Promise Never Resolves

Problem:
  checkCreditsAction() returns a Promise<boolean> that resolves ONLY
  when the user clicks Confirm or Cancel in the modal. If the user:
  (a) navigates away from the page while modal is open
  (b) closes the browser tab
  (c) the modal unmounts due to a parent re-render
  ...the promise NEVER resolves or rejects. Any async function that
  awaits checkCreditsAction() is permanently suspended. The feature
  that called it (e.g., ATS scan, resume enhancement) never executes
  or errors. The user sees a loading spinner indefinitely.

Evidence:
  return new Promise<boolean>((resolve) => {
    setOnConfirm(() => () => {
      closeModals();
      resolve(true);
    });
    setOnCancel(() => () => {
      closeModals();
      resolve(false);
    });
    // No timeout. Promise hangs if neither callback fires.
  });

Solution:
  Add a timeout and cleanup on unmount:

    const CREDIT_MODAL_TIMEOUT = 5 * 60 * 1000; // 5 minutes

    return new Promise<boolean>((resolve) => {
      let resolved = false;

      const safeResolve = (value: boolean) => {
        if (!resolved) {
          resolved = true;
          closeModals();
          resolve(value);
        }
      };

      // Auto-reject if user does nothing for 5 minutes
      const timeout = setTimeout(() => safeResolve(false), CREDIT_MODAL_TIMEOUT);

      setOnConfirm(() => () => {
        clearTimeout(timeout);
        safeResolve(true);
      });

      setOnCancel(() => () => {
        clearTimeout(timeout);
        safeResolve(false);
      });
    });

Impact:
  Without fix — any feature using credit check can get stuck in a
  loading state that never ends. User must hard-refresh to recover.

----------------------------------------------------------------------
Priority  : P2
File      : src/components/layout/Sidebar.tsx
Line      : 78-84
Category  : Bug — Incorrect State Updates After First Interaction

Problem:
  handleToggle is defined as a plain function inside the component.
  The useEffect that registers it as a window event listener has an
  empty dependency array, so it captures the initial reference to
  handleToggle. After the first render, if handleToggle is recreated
  (e.g., after any state update causes a re-render), the event listener
  still holds the OLD reference. The toggle can get out of sync — the
  icon in Header fires the event but Sidebar's state doesn't update.

Evidence:

    window.addEventListener("toggle-sidebar", handleToggle);
    return () => window.removeEventListener("toggle-sidebar", handleToggle);
  }, []); // stale — handleToggle not in deps

Solution:
  const handleToggle = useCallback(() => setIsExpanded((v) => !v), []);
  // useCallback with empty deps is correct here because the functional
  // updater (v) => !v doesn't read external state

  useEffect(() => {
    window.addEventListener("toggle-sidebar", handleToggle);
    return () => window.removeEventListener("toggle-sidebar", handleToggle);
  }, [handleToggle]); // correct dependency

Impact:
  Without fix — sidebar toggle from Header can become unresponsive
  after several state changes, requiring page refresh to fix.
